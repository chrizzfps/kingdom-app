import React, { useState, useEffect } from 'react';
import { 
  Twitter, 
  Instagram, 
  Youtube, 
  Music2, 
  Copy, 
  Check, 
  ExternalLink, 
  Sparkles,
  Upload,
  Image as ImageIcon,
  Loader2,
  AlertCircle,
  Volume2,
  ChevronRight,
  ChevronLeft,
  Wand2,
  Globe,
  LayoutTemplate,
  Plus,
  Briefcase,
  Settings2,
  Users,
  Search,
  ArrowLeft,
  Video,
  Film,
  BookOpen,
  Save,
  PenLine,
  Lightbulb,
  Link as LinkIcon,
  Zap,
  Trash2,
  X
} from 'lucide-react';

const apiKey = ""; // La clave se inyecta automáticamente

const App = () => {
  // --- GLOBAL STATE ---
  const [view, setView] = useState('projects'); // 'projects', 'dashboard', 'wizard'
  
  // Data: Projects
  const [projects, setProjects] = useState([
    { 
      id: 1, 
      name: "Vinos Don Pedro", 
      industry: "Gastronomía", 
      tone: "Elegante, tradicional, evocador", 
      audience: "Amantes del buen vivir, 35-60 años",
      context: "Bodega familiar con 100 años de historia. Enfocados en la calidad y el proceso artesanal.",
      sources: "Web: www.vinosdonpedro.com\nCompetencia: Vinos XYZ\nPalabras clave: Roble, Crianza, Legado, Familia."
    },
    { 
      id: 2, 
      name: "TechMobile Shop", 
      industry: "Tecnología", 
      tone: "Enérgico, directo, técnico pero accesible", 
      audience: "Jóvenes geeks y usuarios buscando ofertas",
      context: "Tienda de reparación y venta de móviles reacondicionados. Precios bajos, garantía alta.",
      sources: "Ofertas de la semana: iPhone 12, Samsung S21.\nUSPs: Garantía 12 meses, Envío 24h."
    }
  ]);
  const [currentProject, setCurrentProject] = useState(null);
  const [projectToDelete, setProjectToDelete] = useState(null); // State for delete modal
  
  // --- DASHBOARD STATE ---
  const [dashboardTab, setDashboardTab] = useState('history'); // 'history', 'identity'
  const [isEditingIdentity, setIsEditingIdentity] = useState(false);
  const [tempIdentity, setTempIdentity] = useState({}); // For editing
  const [isAnalyzingSources, setIsAnalyzingSources] = useState(false);

  // --- WIZARD STATE ---
  const [step, setStep] = useState(1); 
  const [language, setLanguage] = useState('es'); 
  
  // Wizard Data States
  const [mediaFile, setMediaFile] = useState(null);
  const [mediaPreview, setMediaPreview] = useState(null);
  const [videoLink, setVideoLink] = useState(''); // New: External video link
  const [brief, setBrief] = useState('');
  const [aiSuggestions, setAiSuggestions] = useState([]); 
  const [selectedScript, setSelectedScript] = useState('');
  
  // Output States
  const [outputs, setOutputs] = useState({ 
    instagram: '', 
    x: '', 
    tiktok: '', 
    youtube_long: '', 
    youtube_short: '' 
  });
  
  const [youtubeTab, setYoutubeTab] = useState('long'); 
  const [isProcessing, setIsProcessing] = useState(false);
  const [speakingId, setSpeakingId] = useState(null);
  const [copiedId, setCopiedId] = useState(null);

  // --- NEW PROJECT FORM STATE ---
  const [isCreatingProject, setIsCreatingProject] = useState(false);
  const [newProject, setNewProject] = useState({ name: '', tone: '', context: '', audience: '', industry: '', sources: '' });

  // --- CONFIGURATION ---
  const languages = [
    { code: 'es', label: 'Español', flag: '🇪🇸' },
    { code: 'en', label: 'English', flag: '🇺🇸' },
    { code: 'pt', label: 'Português', flag: '🇧🇷' },
  ];

  const platforms = [
    { id: 'instagram', name: 'Instagram', icon: <Instagram className="w-4 h-4" />, url: 'https://instagram.com' },
    { id: 'x', name: 'X / Twitter', icon: <Twitter className="w-4 h-4" />, url: 'https://twitter.com/compose/tweet' },
    { id: 'tiktok', name: 'TikTok', icon: <Music2 className="w-4 h-4" />, url: 'https://tiktok.com/upload' },
    { id: 'youtube', name: 'YouTube', icon: <Youtube className="w-4 h-4" />, url: 'https://studio.youtube.com' }
  ];

  // --- LOGIC: PROJECT MANAGEMENT ---
  const handleCreateProject = () => {
    if (!newProject.name) return;
    const project = { ...newProject, id: Date.now() };
    setProjects([...projects, project]);
    setIsCreatingProject(false);
    setNewProject({ name: '', tone: '', context: '', audience: '', industry: '', sources: '' });
  };

  const selectProject = (project) => {
    setCurrentProject(project);
    setTempIdentity(project); // Initialize edit state
    setDashboardTab('history'); // Default tab
    setView('dashboard');
  };

  const handleDeleteProject = (id) => {
    setProjects(projects.filter(p => p.id !== id));
    setProjectToDelete(null);
  };

  const saveIdentityChanges = () => {
    const updatedProjects = projects.map(p => p.id === tempIdentity.id ? tempIdentity : p);
    setProjects(updatedProjects);
    setCurrentProject(tempIdentity);
    setIsEditingIdentity(false);
  };

  const analyzeBrandSources = async () => {
    if (!tempIdentity.sources) return;
    setIsAnalyzingSources(true);

    const prompt = `Analiza el siguiente texto que contiene fuentes, URLs, notas o información de competencia sobre la marca "${tempIdentity.name}".
    
    FUENTES A ANALIZAR:
    "${tempIdentity.sources}"
    
    TAREA:
    Extrae un "Contexto de Marca" resumido y profesional que defina claramente qué hace la marca, su propuesta de valor única (USP) y sus pilares clave.
    El objetivo es usar este resumen para entrenar a una IA copywriter en el futuro.
    
    Salida: Solo el texto del resumen, sin introducciones.`;

    try {
      const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          contents: [{ parts: [{ text: prompt }] }]
        })
      });

      const data = await response.json();
      const summary = data.candidates[0].content.parts[0].text;
      
      const newContext = tempIdentity.context 
        ? `${tempIdentity.context}\n\n--- ANÁLISIS DE FUENTES ---\n${summary}`
        : summary;

      setTempIdentity({ ...tempIdentity, context: newContext });

    } catch (error) {
      console.error("Error analyzing sources", error);
      alert("Error analizando las fuentes.");
    } finally {
      setIsAnalyzingSources(false);
    }
  };

  const startNewPost = () => {
    setStep(1);
    setMediaFile(null);
    setMediaPreview(null);
    setVideoLink('');
    setBrief('');
    setAiSuggestions([]);
    setSelectedScript('');
    setOutputs({});
    setView('wizard');
  };

  // --- LOGIC: AI GENERATION ---
  
  const generateIdeas = async () => {
    setIsProcessing(true);
    setAiSuggestions([]);
    
    const langName = languages.find(l => l.code === language).label;
    
    // RICH CONTEXT INJECTION
    const projectContext = `
      ESTRATEGIA DE MARCA (BRAND KNOWLEDGE BASE):
      - Cliente: ${currentProject.name} (${currentProject.industry})
      - Tono de Voz (Mandatory): ${currentProject.tone}
      - Audiencia Objetivo: ${currentProject.audience}
      - Contexto General: ${currentProject.context}
      - Fuentes y Recursos Clave: ${currentProject.sources}
    `;

    const prompt = `Actúa como el Director Creativo de la marca descrita en la ESTRATEGIA DE MARCA anterior.
    
    ${projectContext}
    
    TAREA ACTUAL:
    Analiza el material proporcionado.
    ${videoLink ? `URL del Video de Referencia: ${videoLink}` : ''}
    Briefing del usuario: "${brief || 'Genera una idea alineada con la estrategia de marca'}"
    
    INSTRUCCIONES:
    1. Si hay un video/imagen, analízalo visualmente para entender el contenido.
    2. Mantén estrictamente el "Tono de Voz".
    3. Genera 3 ángulos creativos distintos para este post.
    
    Idioma de respuesta: ${langName}.
    
    Formato JSON Array:
    [
      { "title": "Concepto", "content": "Borrador de la idea...", "tone_check": "Cómo cumple con la identidad de marca" },
      ...
    ]`;

    try {
      const payload = {
        contents: [{ parts: [{ text: prompt }] }],
        generationConfig: { responseMimeType: "application/json" }
      };

      // Handle Media (Image or Video)
      if (mediaPreview) {
        const mimeType = mediaFile.type;
        const base64Data = mediaPreview.split(',')[1];
        
        payload.contents[0].parts.push({
          inlineData: { mimeType: mimeType, data: base64Data }
        });
      }

      const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      const data = await response.json();
      const suggestions = JSON.parse(data.candidates[0].content.parts[0].text);
      setAiSuggestions(suggestions);
      setStep(2); 
    } catch (error) {
      console.error(error);
      alert("Error conectando con la IA. Si subiste un video muy pesado, intenta con uno más corto o usa el enlace.");
    } finally {
      setIsProcessing(false);
    }
  };

  const generatePlatformPosts = async () => {
    if (!selectedScript) return;
    setIsProcessing(true);

    const langName = languages.find(l => l.code === language).label;
    
    // ADVANCED PROMPT
    const prompt = `Eres el Community Manager Senior de ${currentProject.name}.
    
    REFERENCIA DE MARCA:
    Tono: ${currentProject.tone}
    Audiencia: ${currentProject.audience}
    
    GUION SELECCIONADO: "${selectedScript}"
    
    Genera los copys finales siguiendo estas reglas TÉCNICAS.
    IMPORTANTE: NO uses negritas (asteriscos **), cursivas ni formato markdown en el texto de salida. Texto plano y emojis solamente.
    
    1. INSTAGRAM: Enfocado a engagement visual. Emojis sutiles. Hashtags estratégicos al final.
    2. X (TWITTER): Tono conversacional. Hilo si es largo (max 280 chars por tweet).
    3. TIKTOK: Descripción optimizada SEO. Incluye Keywords. NO es un guion de video.
    
    4. YOUTUBE LONG (Formato Estructurado):
       Genera un texto con exactamente estas secciones:
       TÍTULO: [Un título clickbait ético]
       
       DESCRIPCIÓN: [Una descripción completa con introducción, puntos clave y CTA]
       
       ETIQUETAS: [Lista de 10 tags separados por comas]

    5. YOUTUBE SHORTS (Formato Estructurado):
       Genera un texto con estas secciones:
       COPY: [Titulo gancho + Breve descripción optimizada]
       
       ETIQUETAS: [Lista de 5 tags #shorts]
    
    Idioma de salida: ${langName}.
    
    JSON: { "instagram": "", "x": "", "tiktok": "", "youtube_long": "", "youtube_short": "" }`;

    try {
      const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          contents: [{ parts: [{ text: prompt }] }],
          generationConfig: { responseMimeType: "application/json" }
        })
      });

      const data = await response.json();
      const results = JSON.parse(data.candidates[0].content.parts[0].text);
      setOutputs(results);
      setStep(3); 
    } catch (error) {
      console.error(error);
    } finally {
      setIsProcessing(false);
    }
  };

  // --- UTILS ---
  const copyToClipboard = (text, id) => {
    navigator.clipboard.writeText(text);
    setCopiedId(id);
    setTimeout(() => setCopiedId(null), 2000);
  };
  
  const handleFileUpload = (e) => {
    const file = e.target.files[0];
    if(file) {
        if (file.size > 20 * 1024 * 1024) {
            alert("El archivo es muy grande para la demo (>20MB). Por favor usa un clip más corto o el enlace de YouTube.");
            return;
        }
        setMediaFile(file);
        const reader = new FileReader();
        reader.onloadend = () => setMediaPreview(reader.result);
        reader.readAsDataURL(file);
    }
  };

  // --- UI COMPONENTS ---

  const renderProjects = () => (
    <div className="max-w-6xl mx-auto px-6 py-12 animate-in fade-in duration-500">
      <div className="flex justify-between items-end mb-12">
        <div>
          <h1 className="text-3xl font-bold text-slate-900 mb-2">Mis Proyectos</h1>
          <p className="text-slate-500">Gestiona la identidad de tus marcas y clientes.</p>
        </div>
        <button 
          onClick={() => setIsCreatingProject(true)}
          className="bg-slate-900 text-white px-6 py-3 rounded-xl font-medium hover:bg-slate-800 transition-all flex items-center gap-2 shadow-lg shadow-slate-200"
        >
          <Plus className="w-5 h-5" /> Nuevo Proyecto
        </button>
      </div>

      {isCreatingProject && (
        <div className="bg-white p-8 rounded-2xl border border-slate-200 shadow-xl mb-12 relative overflow-hidden">
            <h3 className="text-xl font-bold mb-6 flex items-center gap-2"><Briefcase className="w-5 h-5"/> Nuevo Cliente</h3>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-6">
                <div className="space-y-2">
                    <label className="text-sm font-semibold text-slate-700">Nombre</label>
                    <input type="text" value={newProject.name} onChange={e => setNewProject({...newProject, name: e.target.value})} className="w-full p-3 border rounded-lg outline-none focus:ring-2 focus:ring-slate-900" placeholder="Ej. Tesla Motors" />
                </div>
                <div className="space-y-2">
                    <label className="text-sm font-semibold text-slate-700">Industria</label>
                    <input type="text" value={newProject.industry} onChange={e => setNewProject({...newProject, industry: e.target.value})} className="w-full p-3 border rounded-lg outline-none focus:ring-2 focus:ring-slate-900" placeholder="Ej. Automotriz" />
                </div>
                <div className="space-y-2 md:col-span-2">
                    <label className="text-sm font-semibold text-slate-700">Contexto Inicial</label>
                    <textarea value={newProject.context} onChange={e => setNewProject({...newProject, context: e.target.value})} className="w-full p-3 border rounded-lg outline-none h-20 resize-none" placeholder="Breve descripción..." />
                </div>
            </div>
            <div className="flex justify-end gap-3">
                <button onClick={() => setIsCreatingProject(false)} className="px-4 py-2 text-slate-500 hover:bg-slate-50 rounded-lg font-medium">Cancelar</button>
                <button onClick={handleCreateProject} className="px-4 py-2 bg-blue-600 text-white rounded-lg font-medium">Crear</button>
            </div>
        </div>
      )}

      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
        {projects.map(p => (
          <div key={p.id} onClick={() => selectProject(p)} className="group bg-white p-6 rounded-2xl border border-slate-200 hover:border-slate-400 hover:shadow-lg cursor-pointer transition-all flex flex-col h-full relative overflow-hidden">
            <div className="absolute top-0 right-0 p-4 opacity-10 group-hover:opacity-20 transition-opacity">
                <Briefcase className="w-24 h-24" />
            </div>
            <div className="flex justify-between items-start mb-2 z-10 relative">
                 <h3 className="text-xl font-bold text-slate-900">{p.name}</h3>
                 <button 
                    onClick={(e) => { e.stopPropagation(); setProjectToDelete(p); }}
                    className="text-slate-300 hover:text-red-500 transition-colors p-1"
                 >
                    <Trash2 className="w-5 h-5" />
                 </button>
            </div>
            <p className="text-sm text-slate-500 mb-4 font-medium z-10">{p.industry}</p>
            <div className="mt-auto flex items-center gap-2 text-xs text-slate-400 z-10">
                <Settings2 className="w-3 h-3" /> Configurar Identidad
            </div>
          </div>
        ))}
      </div>

      {/* DELETE CONFIRMATION MODAL */}
      {projectToDelete && (
        <div className="fixed inset-0 bg-slate-900/50 backdrop-blur-sm z-[60] flex items-center justify-center p-4">
          <div className="bg-white rounded-2xl p-6 max-w-sm w-full shadow-2xl animate-in fade-in zoom-in-95">
            <h3 className="text-lg font-bold text-slate-900 mb-2">¿Eliminar Proyecto?</h3>
            <p className="text-slate-500 mb-6 text-sm">
              Estás a punto de eliminar <strong>"{projectToDelete.name}"</strong>. Esta acción no se puede deshacer y perderás el historial.
            </p>
            <div className="flex gap-3 justify-end">
              <button 
                onClick={() => setProjectToDelete(null)}
                className="px-4 py-2 text-slate-600 font-medium hover:bg-slate-50 rounded-lg text-sm"
              >
                Cancelar
              </button>
              <button 
                onClick={() => handleDeleteProject(projectToDelete.id)}
                className="px-4 py-2 bg-red-600 text-white font-medium rounded-lg hover:bg-red-700 flex items-center gap-2 text-sm shadow-md shadow-red-100"
              >
                <Trash2 className="w-4 h-4" /> Eliminar Definitivamente
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );

  const renderDashboard = () => (
    <div className="max-w-6xl mx-auto px-6 py-8 animate-in fade-in slide-in-from-right-4">
        {/* Header Navigation */}
        <div className="flex items-center justify-between mb-8">
            <button onClick={() => setView('projects')} className="flex items-center gap-2 text-slate-500 hover:text-slate-900 transition-colors">
                <ArrowLeft className="w-4 h-4" /> Volver a Proyectos
            </button>
            <div className="flex gap-2 bg-slate-100 p-1 rounded-lg">
                <button 
                    onClick={() => setDashboardTab('history')}
                    className={`px-4 py-2 rounded-md text-sm font-medium transition-all ${dashboardTab === 'history' ? 'bg-white shadow text-slate-900' : 'text-slate-500 hover:text-slate-700'}`}
                >
                    Historial
                </button>
                <button 
                    onClick={() => setDashboardTab('identity')}
                    className={`px-4 py-2 rounded-md text-sm font-medium transition-all flex items-center gap-2 ${dashboardTab === 'identity' ? 'bg-white shadow text-slate-900' : 'text-slate-500 hover:text-slate-700'}`}
                >
                    <BookOpen className="w-4 h-4" /> Identidad de Marca
                </button>
            </div>
        </div>

        {/* Project Header Card */}
        <div className="bg-slate-900 text-white rounded-3xl p-8 mb-8 flex flex-col md:flex-row items-center justify-between gap-6 shadow-2xl shadow-slate-200 relative overflow-hidden">
            <div className="absolute top-0 right-0 w-64 h-64 bg-blue-500 rounded-full blur-3xl opacity-10 -translate-y-1/2 translate-x-1/2"></div>
            <div className="z-10">
                <div className="flex items-center gap-3 mb-2">
                    <h1 className="text-3xl font-bold">{currentProject.name}</h1>
                    <span className="bg-slate-700 border border-slate-600 px-3 py-1 rounded-full text-xs font-medium uppercase tracking-wider">{currentProject.industry}</span>
                </div>
                <p className="text-slate-300 max-w-2xl text-sm opacity-80 line-clamp-2">{currentProject.context}</p>
            </div>
            <button 
                onClick={startNewPost}
                className="z-10 bg-white text-slate-900 px-8 py-4 rounded-xl font-bold hover:scale-105 active:scale-95 transition-all flex items-center gap-2 shadow-lg whitespace-nowrap"
            >
                <Sparkles className="w-5 h-5 text-blue-600" /> Crear Nuevo Post
            </button>
        </div>

        {/* TAB: IDENTITY (Brand Settings) */}
        {dashboardTab === 'identity' && (
            <div className="animate-in fade-in slide-in-from-bottom-2">
                <div className="flex justify-between items-center mb-6">
                    <h2 className="text-xl font-bold text-slate-900 flex items-center gap-2">
                        <BookOpen className="w-5 h-5 text-blue-600" /> Knowledge Base
                    </h2>
                    {!isEditingIdentity ? (
                        <button onClick={() => setIsEditingIdentity(true)} className="flex items-center gap-2 text-sm font-medium text-blue-600 hover:text-blue-700 bg-blue-50 px-4 py-2 rounded-lg">
                            <PenLine className="w-4 h-4" /> Editar Identidad
                        </button>
                    ) : (
                        <div className="flex gap-2">
                            <button onClick={() => {setIsEditingIdentity(false); setTempIdentity(currentProject);}} className="text-sm font-medium text-slate-500 hover:text-slate-700 px-4 py-2">Cancelar</button>
                            <button onClick={saveIdentityChanges} className="flex items-center gap-2 text-sm font-medium text-white bg-green-600 hover:bg-green-700 px-4 py-2 rounded-lg shadow-sm">
                                <Save className="w-4 h-4" /> Guardar Cambios
                            </button>
                        </div>
                    )}
                </div>

                <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                    {/* Tone & Audience */}
                    <div className="bg-white p-6 rounded-2xl border border-slate-200 shadow-sm space-y-6">
                        <div>
                            <label className="flex items-center gap-2 text-sm font-bold text-slate-700 mb-2">
                                <Volume2 className="w-4 h-4 text-blue-500" /> Tono de Voz
                            </label>
                            {isEditingIdentity ? (
                                <input 
                                    type="text" 
                                    value={tempIdentity.tone} 
                                    onChange={e => setTempIdentity({...tempIdentity, tone: e.target.value})} 
                                    className="w-full p-3 bg-slate-50 border border-slate-200 rounded-lg outline-none focus:ring-2 focus:ring-blue-500"
                                />
                            ) : (
                                <p className="text-slate-600 bg-slate-50 p-3 rounded-lg">{currentProject.tone}</p>
                            )}
                        </div>
                        <div>
                            <label className="flex items-center gap-2 text-sm font-bold text-slate-700 mb-2">
                                <Users className="w-4 h-4 text-purple-500" /> Audiencia Objetivo
                            </label>
                            {isEditingIdentity ? (
                                <input 
                                    type="text" 
                                    value={tempIdentity.audience} 
                                    onChange={e => setTempIdentity({...tempIdentity, audience: e.target.value})} 
                                    className="w-full p-3 bg-slate-50 border border-slate-200 rounded-lg outline-none focus:ring-2 focus:ring-blue-500"
                                />
                            ) : (
                                <p className="text-slate-600 bg-slate-50 p-3 rounded-lg">{currentProject.audience}</p>
                            )}
                        </div>
                    </div>

                    {/* Detailed Context */}
                    <div className="bg-white p-6 rounded-2xl border border-slate-200 shadow-sm">
                         <label className="flex items-center gap-2 text-sm font-bold text-slate-700 mb-2">
                            <Briefcase className="w-4 h-4 text-orange-500" /> Contexto de Marca
                        </label>
                        {isEditingIdentity ? (
                            <textarea 
                                value={tempIdentity.context} 
                                onChange={e => setTempIdentity({...tempIdentity, context: e.target.value})} 
                                className="w-full p-3 bg-slate-50 border border-slate-200 rounded-lg outline-none focus:ring-2 focus:ring-blue-500 h-32 resize-none"
                            />
                        ) : (
                            <p className="text-slate-600 bg-slate-50 p-3 rounded-lg h-32 overflow-y-auto whitespace-pre-wrap">{currentProject.context}</p>
                        )}
                    </div>

                    {/* Sources & Resources */}
                    <div className="md:col-span-2 bg-white p-6 rounded-2xl border border-slate-200 shadow-sm relative overflow-hidden">
                        <div className="absolute top-0 left-0 w-1 h-full bg-indigo-500"></div>
                        <div className="flex items-center justify-between mb-2">
                            <label className="flex items-center gap-2 text-sm font-bold text-slate-700">
                                <LinkIcon className="w-4 h-4 text-indigo-500"/> Fuentes, Recursos y Notas Adicionales
                            </label>
                            {isEditingIdentity && (
                                <button 
                                    onClick={analyzeBrandSources}
                                    disabled={isAnalyzingSources || !tempIdentity.sources}
                                    className="text-xs bg-indigo-100 text-indigo-700 px-3 py-1.5 rounded-lg font-bold flex items-center gap-1 hover:bg-indigo-200 transition-colors disabled:opacity-50"
                                >
                                    {isAnalyzingSources ? <Loader2 className="w-3 h-3 animate-spin"/> : <Zap className="w-3 h-3"/>}
                                    Analizar y Extraer Identidad
                                </button>
                            )}
                        </div>
                        <p className="text-xs text-slate-400 mb-3">
                            Añade URLs, textos de tu web o info de competencia. Usa el botón "Analizar" para que la IA complete el Contexto automáticamente.
                        </p>
                        {isEditingIdentity ? (
                            <textarea 
                                value={tempIdentity.sources || ''} 
                                onChange={e => setTempIdentity({...tempIdentity, sources: e.target.value})} 
                                className="w-full p-4 bg-slate-50 border border-slate-200 rounded-lg outline-none focus:ring-2 focus:ring-indigo-500 h-40 font-mono text-sm"
                                placeholder="Pega aquí el texto de tu página 'Sobre Nosotros', links de videos de referencia, o notas sobre tu competencia..."
                            />
                        ) : (
                            <div className="bg-slate-50 p-4 rounded-lg h-40 overflow-y-auto font-mono text-sm text-slate-600 whitespace-pre-wrap">
                                {currentProject.sources || <span className="text-slate-400 italic">No hay fuentes definidas. Edita para agregar.</span>}
                            </div>
                        )}
                    </div>
                </div>
            </div>
        )}

        {/* TAB: HISTORY */}
        {dashboardTab === 'history' && (
            <div className="animate-in fade-in slide-in-from-bottom-2">
                <div className="bg-slate-50 border border-dashed border-slate-300 rounded-2xl p-12 text-center text-slate-400">
                    <Search className="w-12 h-12 mx-auto mb-4 opacity-20" />
                    <p className="font-medium">Tu historial de publicaciones para {currentProject.name} aparecerá aquí.</p>
                </div>
            </div>
        )}
    </div>
  );

  const renderWizard = () => (
    <div className="max-w-5xl mx-auto px-6 py-10 animate-in fade-in slide-in-from-bottom-8">
       {/* Wizard Header */}
       <div className="flex items-center justify-between mb-8">
            <button onClick={() => setView('dashboard')} className="flex items-center gap-2 text-sm text-slate-500 hover:text-slate-900">
                <ArrowLeft className="w-4 h-4" /> Cancelar y volver al Dashboard
            </button>
            <div className="flex items-center gap-2">
                <span className="text-xs font-bold uppercase tracking-wider text-slate-400">Proyecto Activo:</span>
                <span className="text-sm font-bold bg-indigo-50 text-indigo-700 border border-indigo-100 px-3 py-1 rounded-full">{currentProject.name}</span>
            </div>
       </div>

       <div className="flex items-center justify-center space-x-4 mb-12">
            {[1, 2, 3].map((s) => (
                <div key={s} className="flex items-center">
                <div className={`w-8 h-8 rounded-full flex items-center justify-center text-xs font-bold transition-all duration-300 ${step === s ? 'bg-slate-900 text-white ring-4 ring-slate-100' : step > s ? 'bg-green-500 text-white' : 'bg-slate-100 text-slate-400'}`}>
                    {step > s ? <Check className="w-4 h-4" /> : s}
                </div>
                {s < 3 && <div className={`w-12 h-0.5 mx-2 ${step > s ? 'bg-green-500' : 'bg-slate-200'}`} />}
                </div>
            ))}
       </div>

       {step === 1 && (
         <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
            <div className="space-y-4">
                 {/* Media Uploader (Video & Image) */}
                <div onClick={() => document.getElementById('file-upload').click()} className={`group relative aspect-video rounded-2xl border-2 border-dashed flex flex-col items-center justify-center cursor-pointer transition-all duration-300 ${mediaPreview ? 'border-transparent shadow-xl' : 'border-slate-200 hover:border-blue-400 bg-slate-50'}`}>
                    <input id="file-upload" type="file" className="hidden" onChange={handleFileUpload} accept="image/*,video/*" />
                    {mediaPreview ? (
                        mediaFile?.type.startsWith('video') ? 
                        <video src={mediaPreview} className="w-full h-full object-cover rounded-2xl" controls /> :
                        <img src={mediaPreview} className="w-full h-full object-cover rounded-2xl" alt="Preview" />
                    ) : (
                        <div className="text-center p-6">
                            <div className="flex justify-center gap-2 mb-3 text-slate-300">
                                <ImageIcon className="w-8 h-8" />
                                <Film className="w-8 h-8" />
                            </div>
                            <p className="font-semibold text-slate-600">Sube Foto o Video</p>
                            <p className="text-xs text-slate-400 mt-1">Soporta clips cortos para análisis visual</p>
                        </div>
                    )}
                </div>
                
                {/* External Link Input */}
                <div className="flex items-center gap-2 bg-white border border-slate-200 rounded-xl p-3 shadow-sm">
                    <LinkIcon className="w-4 h-4 text-slate-400" />
                    <input 
                        type="text" 
                        value={videoLink}
                        onChange={(e) => setVideoLink(e.target.value)}
                        placeholder="O pega un enlace de YouTube / Video aquí"
                        className="flex-grow outline-none text-sm text-slate-700"
                    />
                </div>
            </div>

            <div className="flex flex-col justify-center space-y-4">
                <div className="bg-indigo-50 p-4 rounded-xl border border-indigo-100 text-sm text-indigo-900 mb-2">
                    <div className="flex items-center gap-2 font-bold mb-1"><Sparkles className="w-4 h-4" /> Brain Active</div>
                    La IA analizará este contenido usando el tono <strong>"{currentProject.tone}"</strong> y consultará tus fuentes guardadas.
                </div>
                <textarea 
                    value={brief}
                    onChange={(e) => setBrief(e.target.value)}
                    className="w-full h-40 p-4 rounded-xl border border-slate-200 focus:ring-2 focus:ring-slate-900 outline-none resize-none"
                    placeholder={`¿Qué quieres comunicar hoy para ${currentProject.name}?`}
                />
                <button 
                  onClick={generateIdeas}
                  disabled={isProcessing}
                  className="w-full py-4 bg-slate-900 text-white rounded-xl font-bold hover:bg-slate-800 transition-all flex items-center justify-center gap-2 disabled:opacity-50"
                >
                  {isProcessing ? <Loader2 className="w-5 h-5 animate-spin" /> : <Wand2 className="w-5 h-5" />}
                  Generar Conceptos On-Brand
                </button>
            </div>
         </div>
       )}

       {step === 2 && (
         <div className="animate-in fade-in">
            <h2 className="text-2xl font-bold mb-6 text-center">Enfoques Estratégicos Sugeridos</h2>
            <div className="grid grid-cols-1 md:grid-cols-3 gap-6 mb-8">
              {aiSuggestions.map((suggestion, idx) => (
                <div key={idx} onClick={() => setSelectedScript(suggestion.content)} className={`p-6 rounded-2xl border-2 cursor-pointer transition-all hover:scale-[1.02] ${selectedScript === suggestion.content ? 'border-slate-900 bg-slate-50 shadow-md' : 'border-slate-100 bg-white hover:border-slate-300'}`}>
                  <h3 className="font-bold text-lg mb-2">{suggestion.title}</h3>
                  <p className="text-sm text-slate-600 mb-4">{suggestion.content}</p>
                  <div className="text-xs bg-slate-100 p-2 rounded text-slate-500 italic">✅ {suggestion.tone_check}</div>
                </div>
              ))}
            </div>
            <div className="flex justify-center">
                <button onClick={generatePlatformPosts} disabled={!selectedScript || isProcessing} className="px-10 py-3 bg-slate-900 text-white rounded-full font-bold hover:bg-slate-800 transition-all flex items-center gap-2 disabled:opacity-50">
                    {isProcessing ? <Loader2 className="w-4 h-4 animate-spin" /> : "Crear Copys Definitivos"} <ChevronRight className="w-4 h-4" />
                </button>
            </div>
         </div>
       )}

       {step === 3 && (
         <div className="grid grid-cols-1 md:grid-cols-2 gap-6 pb-20 animate-in fade-in">
            {platforms.map((platform) => {
                const isYoutube = platform.id === 'youtube';
                const content = isYoutube ? (youtubeTab === 'long' ? outputs.youtube_long : outputs.youtube_short) : outputs[platform.id];
                return (
                    <div key={platform.id} className="bg-white rounded-2xl border border-slate-200 shadow-sm overflow-hidden flex flex-col">
                        <div className="px-6 py-4 border-b border-slate-50 bg-slate-50/50 flex items-center justify-between">
                            <div className="flex items-center gap-3">
                                <div className="p-2 bg-white rounded-lg border border-slate-100 shadow-sm">{platform.icon}</div>
                                <span className="font-bold text-slate-900">{platform.name}</span>
                            </div>
                            {isYoutube && (
                                <div className="flex bg-slate-200 rounded-lg p-1">
                                    <button onClick={() => setYoutubeTab('long')} className={`px-3 py-1 rounded-md text-xs font-bold transition-all ${youtubeTab === 'long' ? 'bg-white shadow text-slate-900' : 'text-slate-500'}`}>Video</button>
                                    <button onClick={() => setYoutubeTab('short')} className={`px-3 py-1 rounded-md text-xs font-bold transition-all ${youtubeTab === 'short' ? 'bg-white shadow text-slate-900' : 'text-slate-500'}`}>Shorts</button>
                                </div>
                            )}
                        </div>
                        <div className="p-6 flex-grow">
                             <textarea value={content} onChange={(e) => { if(isYoutube) { setOutputs({...outputs, [youtubeTab === 'long' ? 'youtube_long' : 'youtube_short']: e.target.value}); } else { setOutputs({...outputs, [platform.id]: e.target.value}); }}} className="w-full h-40 bg-transparent border-none p-0 focus:ring-0 text-sm text-slate-600 resize-none leading-relaxed font-mono" />
                        </div>
                        <div className="px-6 py-4 bg-slate-50 border-t border-slate-100 flex gap-3">
                            <button onClick={() => copyToClipboard(content, platform.id)} className="flex-1 py-2 bg-white border border-slate-200 rounded-lg text-xs font-bold text-slate-700 hover:bg-slate-100 transition-all flex items-center justify-center gap-2">
                                {copiedId === platform.id ? <Check className="w-3 h-3 text-green-500" /> : <Copy className="w-3 h-3" />} Copiar
                            </button>
                        </div>
                    </div>
                );
            })}
         </div>
       )}
    </div>
  );

  return (
    <div className="min-h-screen bg-white text-slate-900 font-sans selection:bg-slate-100">
      <nav className="border-b border-slate-100 sticky top-0 bg-white/80 backdrop-blur-md z-50">
        <div className="max-w-6xl mx-auto px-6 h-16 flex items-center justify-between">
          <div className="flex items-center gap-2 cursor-pointer" onClick={() => setView('projects')}>
            <div className="w-8 h-8 bg-slate-900 rounded-lg flex items-center justify-center text-white"><LayoutTemplate className="w-4 h-4" /></div>
            <span className="font-bold text-lg tracking-tight">SocialCenter<span className="text-slate-400 font-normal">.ai</span></span>
          </div>
          <div className="flex items-center gap-4">
             <div className="relative group">
              <button className="flex items-center gap-2 text-sm font-medium text-slate-600 hover:text-slate-900 transition-colors"><Globe className="w-4 h-4" /> {languages.find(l => l.code === language).code.toUpperCase()}</button>
              <div className="absolute right-0 top-full mt-2 w-32 bg-white border border-slate-100 rounded-lg shadow-lg opacity-0 invisible group-hover:opacity-100 group-hover:visible transition-all transform origin-top-right">
                {languages.map(l => (<button key={l.code} onClick={() => setLanguage(l.code)} className="w-full text-left px-4 py-2 text-sm hover:bg-slate-50">{l.label} {l.flag}</button>))}
              </div>
            </div>
          </div>
        </div>
      </nav>
      <main>
        {view === 'projects' && renderProjects()}
        {view === 'dashboard' && currentProject && renderDashboard()}
        {view === 'wizard' && currentProject && renderWizard()}
      </main>
    </div>
  );
};

export default App;
